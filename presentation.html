// authService.js
export async function createInvitationToken({ email, role }) {
  const jti = uuidv4();

  // Rôle prédéfini — enregistré avec le token
  await createInvitationModel({ jti, email, role });

  // Lien unique — expire dans 48h
  return jwt.sign(
    { email, role, jti, purpose: 'invitation' },
    process.env.JWT_SECRET,
    { expiresIn: '48h' }
  );
}
// authService.js
export async function decodeInvitationToken(token) {
  const decoded = jwt.verify(token, JWT_SECRET);

  if (decoded.purpose !== 'invitation') {
    throw new Error('Token invalide');
  }

  // Vérification que le lien n'a pas déjà été utilisé
  const invitation = await getInvitationByJtiModel(decoded.jti);
  if (!invitation || invitation.status !== 'pending') {
    throw new Error('Lien déjà utilisé ou expiré');
  }

  return { email: decoded.email, role: decoded.role, jti: decoded.jti };
}
// authService.js
export async function register({ token, firstname, lastname, password }) {
  // Vérification du token et récupération du rôle prédéfini
  const decoded = await decodeInvitationToken(token);

  // Hachage du mot de passe
  const hash = await bcrypt.hash(password, 10);

  // Création du compte avec le rôle défini par l'admin
  const userId = await createUserModel({
    email: decoded.email,
    firstname,
    lastname,
    password_hash: hash,
    role_id: roleData.id
  });

  // Token invalidé — usage unique garanti
  await markInvitationAsUsedModel(decoded.jti);

  // Email de bienvenue
  await welcomeEmail(decoded.email, firstname);

  return { id: userId, email: decoded.email };
}
